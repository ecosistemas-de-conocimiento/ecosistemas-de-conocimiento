# Model-server dependencies, shared verbatim by rerank-service and bge-m3-service.
# The two CPU siblings stay in lockstep on one FlagEmbedding / torch / transformers set.
#
# 🔴 THESE TWO FILES MUST BE BYTE-IDENTICAL, INCLUDING THIS COMMENT, AND THAT IS WHY
# IT NAMES NEITHER SERVICE. Both Dockerfiles are instruction-identical, so Docker
# reuses the ~1.61 GB pip layer across both images — but only while the COPY'd
# requirements.txt hashes the same. Earlier each file's first line named
# its OWN service while claiming byte-identity, which was the only difference between
# them, and it split the layer: `docker system df -v` showed just 134.4 MB shared and
# the deps layer stored TWICE. A single word of drift here costs 1.61 GB per host.
#
# torch is installed FIRST from the CPU wheel index in the Dockerfile (so no CUDA
# is ever resolved); it is also listed here so the version is pinned in one place.
#
# The transitive deps below are the MODEL-LOAD-CRITICAL set. They are pinned
# explicitly because FlagEmbedding==1.3.4 only floor-pins them (transformers>=…),
# so a rebuild months from now would otherwise resolve a newer, possibly
# incompatible transformers/tokenizers and break at model-load. For a fully
# reproducible image, capture a complete `pip freeze` lock from the first green
# build and commit it as requirements.lock (see README → "Dependency pinning").

# Web layer
fastapi==0.111.0
uvicorn[standard]==0.30.1

# Model runtime. FlagEmbedding 1.3.x exposes FlagReranker(..., use_fp16=...) /
# BGEM3FlagModel(..., use_fp16=...) and the compute_score / encode signatures the
# services call.
FlagEmbedding==1.3.4

# CPU torch (installed from the CPU wheel index in the Dockerfile — no CUDA).
# 2.4.1 has cp311 CPU wheels and works with the FlagEmbedding pin above.
torch==2.4.1

# ── Load-critical transitive pins (see header) ────────────────────────────────
transformers==4.44.2
tokenizers==0.19.1
huggingface-hub==0.24.6
safetensors==0.4.5
sentence-transformers==3.0.1
# Pin numpy to the 1.x ABI: the whole 2024-era torch/transformers/scipy set here
# predates the numpy-2 ABI churn; 1.26.4 avoids import-time ABI breakage.
numpy==1.26.4
